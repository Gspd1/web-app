SELECT
    recipe.name AS recipe_name,
    recipe_step.recipe_step_number,
    recipe_step.recipe_step_image
FROM recipe_step
JOIN recipe ON recipe_step.recipe_id = recipe.id
WHERE recipe.name IN ('Куриный суп', 'Шоколадный кекс')
ORDER BY recipe.name, recipe_step.recipe_step_number;
